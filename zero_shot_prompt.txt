function toSnakeCase(text) {
  return text
    .replace(/\s+/g, ' ')                // Normalize whitespace
    .trim()                              // Remove leading/trailing spaces
    .replace(/([a-z])([A-Z])/g, '$1_$2') // Handle camelCase
    .replace(/[\s\-]+/g, '_')            // Replace spaces and hyphens with _
    .toLowerCase();
}
